local function playSound(id, vol, loop)
    local s = Instance.new("Sound")
    s.SoundId = id
    s.Volume = vol or 0.4
    s.Looped = loop or false
    s.Parent = screenGui
    s:Play()
    s.Ended:Connect(function() if not s.Looped then s:Destroy() end end)
    return s
end

local function applyHackerStyle(obj, strokeColor, transparency)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 4)
    corner.Parent = obj
    local stroke = Instance.new("UIStroke")
    stroke.Color = strokeColor or Color3.fromRGB(0, 255, 100)
    stroke.Thickness = 1.2
    stroke.Transparency = transparency or 0.4
    stroke.Parent = obj
    return stroke
end

local function createPopGlitch()
    local colors = {Color3.fromRGB(255,0,80), Color3.fromRGB(0,255,255), Color3.fromRGB(0,255,100)}
    local pop = Instance.new("Frame", screenGui)
    pop.Size = UDim2.new(math.random(0, 1), math.random(50, 400), 0, math.random(1, 5))
    pop.Position = UDim2.new(math.random(), -200, math.random(), 0)
    pop.BackgroundColor3 = colors[math.random(1, #colors)]
    pop.BorderSizePixel = 0
    pop.ZIndex = 3000
    task.delay(0.05, function() pop:Destroy() end)
end

local function intenseGlitch(times)
    local originalPos = mainFrame.Position
    playSound(SOUNDS.Static, 0.3)
    for i = 1, times do
        createPopGlitch()
        local x = math.random(-35, 35)
        local y = math.random(-35, 35)
        mainFrame.Position = originalPos + UDim2.new(0, x, 0, y)
        mainFrame.Rotation = math.random(-5, 5)
        mainFrame.BackgroundColor3 = (i % 2 == 0) and Color3.fromRGB(20, 20, 20) or Color3.fromRGB(0,0,0)
        task.wait(0.02)
    end
    mainFrame.Position = originalPos
    mainFrame.Rotation = 0
    mainFrame.BackgroundColor3 = Color3.fromRGB(0,0,0)
end

local function animateUI(visible)
    if isAnimating then return end
    isAnimating = true
    if visible then
        playSound(SOUNDS.Toggle, 0.5) -- Son identique à la touche L
        mainFrame.Visible = true
        intenseGlitch(15)
        uiVisible = true
        notify("ZOKA_OS_READY")
    else
        intenseGlitch(8)
        mainFrame.Visible = false
        uiVisible = false
    end
    isAnimating = false
end

local function updateUIList()
    local list = mainFrame:FindFirstChild("DataScroll") or Instance.new("ScrollingFrame")
    if not list.Parent then
        list.Name = "DataScroll"
        list.Size = UDim2.new(0, 340, 0, 280)
        list.Position = UDim2.new(1.05, 10, 0, 0)
        list.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
        list.Visible = false
        list.Parent = mainFrame
        applyHackerStyle(list, Color3.fromRGB(255, 0, 100), 0.3)
        Instance.new("UIListLayout", list).Padding = UDim.new(0, 10)
    end
    for _, v in pairs(list:GetChildren()) do if v:IsA("Frame") then v:Destroy() end end
    for i, data in pairs(savedPoints) do
        local row = Instance.new("Frame", list)
        row.Size = UDim2.new(0.95, 0, 0, 50)
        row.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
        applyHackerStyle(row, Color3.fromRGB(0, 255, 100), 0.7)
        local input = Instance.new("TextBox", row)
        input.Size = UDim2.new(0.5, 0, 1, 0)
        input.Position = UDim2.new(0.05, 0, 0, 0)
        input.BackgroundTransparency = 1
        input.Text = data.name
        input.TextColor3 = Color3.fromRGB(255, 255, 255)
        input.Font = Enum.Font.Code
        input.TextSize = 14
        input.FocusLost:Connect(function() data.name = input.Text end)
        
        local tp = createHackerBtn("JUMP", UDim2.new(0.58, 0, 0.2, 0), UDim2.new(0.18, 0, 0.6, 0))
        tp.Parent = row
        tp.MouseButton1Click:Connect(function()
            if player.Character:FindFirstChild("HumanoidRootPart") then
                playSound(SOUNDS.Teleport, 0.6)
                player.Character.HumanoidRootPart.CFrame = CFrame.new(data.pos) + Vector3.new(0,3,0)
                notify("NODE_REACHED")
            end
        end)
        
        local del = createHackerBtn("X", UDim2.new(0.78, 0, 0.2, 0), UDim2.new(0.18, 0, 0.6, 0), Color3.fromRGB(255, 0, 80))
        del.Parent = row
        del.MouseButton1Click:Connect(function() 
            playSound(SOUNDS.Delete, 0.6)
            table.remove(savedPoints, i) 
            updateUIList() 
        end)
    end
end
